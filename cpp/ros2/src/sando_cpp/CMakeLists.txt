cmake_minimum_required(VERSION 3.8)
project(sando_cpp)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(dynus_interfaces REQUIRED)

# the golden-verified algorithm headers + vendored Eigen/LBFGSpp (header-only)
set(ALGO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../..)   # -> cpp/
include_directories(${ALGO_ROOT}/include ${ALGO_ROOT}/third_party/eigen ${ALGO_ROOT}/third_party)

add_executable(sando_node src/sando_node.cpp)
ament_target_dependencies(sando_node rclcpp std_msgs geometry_msgs nav_msgs dynus_interfaces)
add_executable(fake_sim src/fake_sim.cpp)
ament_target_dependencies(fake_sim rclcpp std_msgs geometry_msgs nav_msgs tf2_ros dynus_interfaces)
add_executable(perclass_obstacle_pub src/perclass_obstacle_pub.cpp)
ament_target_dependencies(perclass_obstacle_pub rclcpp dynus_interfaces)
add_executable(auto_goal_pub src/auto_goal_pub.cpp)
ament_target_dependencies(auto_goal_pub rclcpp geometry_msgs)
add_executable(convert_odom_to_state src/convert_odom_to_state.cpp)
ament_target_dependencies(convert_odom_to_state rclcpp nav_msgs dynus_interfaces)
add_executable(convert_vicon_to_state src/convert_vicon_to_state.cpp)
ament_target_dependencies(convert_vicon_to_state rclcpp geometry_msgs dynus_interfaces)
add_executable(convert_goal_to_cmd_vel src/convert_goal_to_cmd_vel.cpp)
ament_target_dependencies(convert_goal_to_cmd_vel rclcpp geometry_msgs dynus_interfaces)
add_executable(odom_to_global_state src/odom_to_global_state.cpp)
ament_target_dependencies(odom_to_global_state rclcpp nav_msgs geometry_msgs tf2_ros dynus_interfaces)
add_executable(obstacle_tracker src/obstacle_tracker.cpp)
ament_target_dependencies(obstacle_tracker rclcpp sensor_msgs visualization_msgs std_msgs geometry_msgs dynus_interfaces)

install(TARGETS sando_node fake_sim perclass_obstacle_pub auto_goal_pub
        convert_odom_to_state convert_vicon_to_state convert_goal_to_cmd_vel odom_to_global_state
        obstacle_tracker
        DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
